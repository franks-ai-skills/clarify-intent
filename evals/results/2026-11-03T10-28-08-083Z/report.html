```python
from datetime import date, datetime

# Formats to try in order after ISO 8601. If a date is ambiguous, the first
# format that matches wins. Numeric dates like 03/04/2025 are read
# month-first (US style).
_FORMATS = (
    "%Y-%m-%d",    # 2025-03-04
    "%Y/%m/%d",    # 2025/03/04
    "%Y.%m.%d",    # 2025.03.04
    "%Y%m%d",      # 20250304
    "%m/%d/%Y",    # 03/04/2025
    "%m-%d-%Y",    # 03-04-2025
    "%d %B %Y",    # 4 March 2025
    "%d %b %Y",    # 4 Mar 2025
    "%B %d %Y",    # March 4 2025
    "%b %d %Y",    # Mar 4 2025
    "%d-%b-%Y",    # 04-Mar-2025
)


def parse_date(value: str) -> date:
    """Parse a date string into a ``datetime.date``.

    Tries ISO 8601 first (including full datetimes such as
    ``2025-03-04T10:15:00Z``, whose time part is dropped), then a set of
    common formats. Leading and trailing spaces are ignored. Commas are
    ignored too, so "March 4, 2025" works. Month names are matched
    case-insensitively.

    Raises:
        TypeError: if ``value`` is not a string.
        ValueError: if the string doesn't match any supported format
            or isn't a real calendar date (e.g. 2025-02-30).
    """
    if not isinstance(value, str):
        raise TypeError(f"expected str, got {type(value).__name__}")

    s = value.strip()
    if not s:
        raise ValueError("empty date string")

    # Fast path: ISO 8601 date or datetime.
    try:
        return date.fromisoformat(s)
    except ValueError:
        pass
    try:
        return datetime.fromisoformat(s.replace("Z", "+00:00")).date()
    except ValueError:
        pass

    normalized = " ".join(s.replace(",", " ").split())
    for fmt in _FORMATS:
        try:
            return datetime.strptime(normalized, fmt).date()
        except ValueError:
            continue

    raise ValueError(f"unrecognized date format: {value!r}")
```

Example usage:

```python
parse_date("2025-03-04")            # date(2025, 3, 4)
parse_date("2025-03-04T10:15:00Z")  # date(2025, 3, 4)
parse_date("March 4, 2025")         # date(2025, 3, 4)
parse_date("4 Mar 2025")            # date(2025, 3, 4)
parse_date("03/04/2025")            # date(2025, 3, 4)  (month first)
parse_date("2025-02-30")            # ValueError
```

**Choices I made:**
- **Standard library only:** I didn't use `dateutil`, so there are no extra dependencies. The trade-off is that only the listed formats are accepted, which keeps the results predictable.
- **ISO 8601 is tried first**, because it's unambiguous and the most common format in data. For full datetimes, the time and time zone are dropped.
- **Numeric dates are read month-first:** `03/04/2025` becomes March 4. If you want day-first, swap in `%d/%m/%Y` and `%d-%m-%Y` for the two month-first lines.
- **Errors:** Unparseable input and impossible dates raise `ValueError`, and anything that isn't a string raises `TypeError`. The function never returns `None`.
- **Month names follow the system locale:** `strptime` reads them in the current locale, which is English by default.

```python
    "%d.%m.%Y",      # 04.03.2025
    "%B %d, %Y",     # March 4, 2025
    "%b %d, %Y",     # Mar 4, 2025
    "%B %d %Y",      # March 4 2025
    "%b %d %Y",      # Mar 4 2025

def parse_date(value: str) -> date:
    """Parse a date string into a datetime.date.

    Tries ISO 8601 first (including full datetimes such as
    '2025-03-04T10:30:00Z', whose time part is dropped), then a set of
    common formats. Month and day names are case-insensitive.
    Ambiguous numeric dates are read month-first.

    Raises:
        TypeError: if value is not a str.
        ValueError: if the string matches no supported format.
    """
    if not isinstance(value, str):
        raise TypeError(f"expected str, got {type(value).__name__}")

    s = " ".join(value.split())  # trim and collapse whitespace
    if not s:
        raise ValueError("empty date string")

    # Fast path: ISO 8601 date or datetime.
    try:
        return date.fromisoformat(s)
    except ValueError:
        pass
    try:
        return datetime.fromisoformat(s.replace("Z", "+00:00")).date()
    except ValueError:
        pass

    for fmt in _FORMATS:
        try:
            return datetime.strptime(s, fmt).date()
        except ValueError:
            continue

    raise ValueError(f"unrecognized date format: {value!r}")
```
